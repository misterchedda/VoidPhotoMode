cmake_minimum_required(VERSION 3.20)
project(VoidPhotoMode VERSION 0.8.0 LANGUAGES C CXX)
if(NOT MSVC)
  message(FATAL_ERROR "The supported toolchain is MSVC (Visual Studio 2022)")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(MH third_party/MinHook)
add_library(minhook STATIC ${MH}/src/buffer.c ${MH}/src/hook.c ${MH}/src/trampoline.c ${MH}/src/hde/hde64.c)
target_include_directories(minhook PUBLIC ${MH}/include)
add_library(VoidPhotoBackend SHARED src/core/backend.cpp src/input/input_capture.cpp src/ui/native_ui.cpp src/effects/post_fx.cpp src/effects/color_fx.cpp src/capture/photo_capture.cpp src/capture/capture_io.cpp src/core/runtime.cpp src/ui/hud_patch.cpp)
target_compile_features(VoidPhotoBackend PRIVATE cxx_std_17)
target_include_directories(VoidPhotoBackend PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_definitions(VoidPhotoBackend PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(VoidPhotoBackend PRIVATE minhook user32 dinput8 dxguid d3d11 d3dcompiler windowscodecs ole32 shell32 bcrypt)
target_compile_options(VoidPhotoBackend PRIVATE /W4 /EHsc)
set(PACKAGE_DIR "${CMAKE_BINARY_DIR}/package")
set_target_properties(VoidPhotoBackend PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${PACKAGE_DIR}/VoidPhotoMode")
add_library(dinput8 SHARED src/loader/proxy.cpp src/loader/proxy.def)
target_compile_features(dinput8 PRIVATE cxx_std_17)
target_compile_definitions(dinput8 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(dinput8 PRIVATE /W4 /EHsc)
set_target_properties(dinput8 PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${PACKAGE_DIR}" PREFIX "")
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND)
    add_test(NAME ui_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_contract.py --check)
  endif()
  add_executable(proxy_test tests/proxy_test.cpp)
  target_link_libraries(proxy_test PRIVATE dxguid)
  add_test(NAME proxy_forwarding COMMAND proxy_test "$<TARGET_FILE:dinput8>")
  add_executable(hud_patch_test tests/hud_patch_test.cpp src/ui/hud_patch.cpp)
  target_compile_features(hud_patch_test PRIVATE cxx_std_17)
  target_include_directories(hud_patch_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  add_test(NAME hud_patch_bounds COMMAND hud_patch_test)
  set(HUD_ORIGINAL "" CACHE FILEPATH "Local original HUD fixture, optional")
  set(HUD_PATCHED "" CACHE FILEPATH "Local patched HUD fixture, optional")
  if(HUD_ORIGINAL AND HUD_PATCHED)
    add_test(NAME hud_patch_exact COMMAND hud_patch_test "${HUD_ORIGINAL}" "${HUD_PATCHED}")
  endif()
  add_executable(capture_test tests/capture_test.cpp src/capture/capture_io.cpp)
  target_compile_features(capture_test PRIVATE cxx_std_17)
  target_compile_definitions(capture_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_include_directories(capture_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  target_link_libraries(capture_test PRIVATE d3d11 windowscodecs ole32 shell32 bcrypt)
  add_test(NAME capture_png COMMAND capture_test)
  add_executable(color_gpu_test tests/color_gpu_test.cpp src/effects/color_fx.cpp)
  target_compile_features(color_gpu_test PRIVATE cxx_std_17)
  target_compile_definitions(color_gpu_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_include_directories(color_gpu_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  target_link_libraries(color_gpu_test PRIVATE minhook d3d11 d3dcompiler)
  add_test(NAME color_gpu COMMAND color_gpu_test)
  add_test(NAME color_gpu_deferred COMMAND color_gpu_test deferred)
  add_executable(pointer_controls_test tests/pointer_controls_test.cpp)
  target_compile_features(pointer_controls_test PRIVATE cxx_std_17)
  target_include_directories(pointer_controls_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  add_test(NAME pointer_controls COMMAND pointer_controls_test)
  add_executable(camera_controls_test tests/camera_controls_test.cpp)
  target_compile_features(camera_controls_test PRIVATE cxx_std_17)
  target_include_directories(camera_controls_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  add_test(NAME camera_controls COMMAND camera_controls_test)
  add_executable(focus_controls_test tests/focus_controls_test.cpp)
  target_compile_features(focus_controls_test PRIVATE cxx_std_17)
  target_include_directories(focus_controls_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  add_test(NAME focus_controls COMMAND focus_controls_test)
  add_executable(discrete_input_test tests/discrete_input_test.cpp)
  target_compile_features(discrete_input_test PRIVATE cxx_std_17)
  target_include_directories(discrete_input_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  add_test(NAME discrete_input COMMAND discrete_input_test)
endif()

install(TARGETS dinput8 RUNTIME DESTINATION .)
install(TARGETS VoidPhotoBackend RUNTIME DESTINATION VoidPhotoMode)
install(FILES docs/INSTALL.txt DESTINATION VoidPhotoMode RENAME README.txt)
install(FILES LICENSE THIRD_PARTY_NOTICES.txt DESTINATION VoidPhotoMode)
install(FILES third_party/MinHook/LICENSE.txt DESTINATION VoidPhotoMode/third_party/MinHook)
